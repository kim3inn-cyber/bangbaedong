      const desc = (text || '').toLowerCase();
      for (const item of emojiMap) {
        if (item.keywords.some(keyword => desc.includes(keyword))) return item.emoji;
      }
      return '🏬';

    function App() {
      const [shops, setShops] = useState([]);
      const [shopName, setShopName] = useState('');
      const [shopDesc, setShopDesc] = useState('');
      const [shopProfit, setShopProfit] = useState('');
      const [isReadyToPlace, setIsReadyToPlace] = useState(false);
      const [isCompleted, setIsCompleted] = useState(false);
      const [toastMessage, setToastMessage] = useState('');
      const [showConfirmReset, setShowConfirmReset] = useState(false);
      const mapRef = useRef(null);

      // 우리집이 기본 상점이 아니므로 배열 전체 길이를 사용합니다.
      const shopCount = shops.length;
      const previewEmoji = findEmoji(shopName + ' ' + shopDesc);

      const handleReady = (e) => {
        e.preventDefault();
        if (!shopName || !shopDesc || !shopProfit) {
          setToastMessage('빈칸을 모두 채워주세요! ✍️');
          setTimeout(() => setToastMessage(''), 3000);
          return;
        }
        setIsReadyToPlace(true);
      };

      const handleMapClick = (e) => {
        if (!isReadyToPlace || isCompleted) return;

        const rect = mapRef.current.getBoundingClientRect();
        const xPercent = Math.max(5, Math.min(95, ((e.clientX - rect.left) / rect.width) * 100));
        const yPercent = Math.max(5, Math.min(95, ((e.clientY - rect.top) / rect.height) * 100));

        const newShop = {
          id: Date.now(),
          x: xPercent,
          y: yPercent,
          name: shopName,
          desc: shopDesc,
          icon: previewEmoji, 
          profit: shopProfit,
          isBouncing: true
        };

        setShops([...shops, newShop]);
        setShopName('');
        setShopDesc('');
        setShopProfit('');
        setIsReadyToPlace(false);

        setTimeout(() => {
          setShops(currentShops => 
            currentShops.map(s => s.id === newShop.id ? { ...s, isBouncing: false } : s)
          );
        }, 1000);
      };

      const handleQuickFill = (name, desc, profit) => {
        setShopName(name);
        setShopDesc(desc);
        setShopProfit(profit);
      };

      return (
        <div className={`min-h-screen flex flex-col p-4 md:p-6 transition-colors duration-1000 ${isCompleted ? 'bg-indigo-950 text-white' : 'bg-amber-50/50 text-slate-800'}`}>
          
          {toastMessage && (
            <div className="fixed top-6 left-1/2 transform -translate-x-1/2 z-[100] bg-gray-900 text-white px-6 py-3 rounded-full font-bold shadow-2xl animate-bounce border-2 border-white">
              {toastMessage}
            </div>
          )}

          {showConfirmReset && (
            <div className="fixed inset-0 z-[100] flex items-center justify-center bg-black/60 backdrop-blur-sm">
              <div className="bg-white text-slate-800 p-8 rounded-3xl shadow-2xl text-center border-4 border-pink-300 max-w-sm mx-4">
                <div className="text-5xl mb-4">🧹</div>
                <h3 className="text-xl font-bold mb-6">마을 지도를 처음부터<br/>다시 만들까요?</h3>
                <div className="flex justify-center gap-3">
                  <button onClick={() => { setShops([]); setIsCompleted(false); setShowConfirmReset(false); }} className="px-5 py-3 bg-pink-500 text-white rounded-xl font-bold hover:bg-pink-600 transition-colors">네, 지울래요</button>
                  <button onClick={() => setShowConfirmReset(false)} className="px-5 py-3 bg-gray-200 text-gray-700 rounded-xl font-bold hover:bg-gray-300 transition-colors">아니요</button>
                </div>

            </div>

                  <button type="button" onClick={() => handleQuickFill('초등 알파 문구', '연필 공책 문구점', '3만원')} className="bg-amber-100 hover:bg-amber-200 text-amber-800 px-2 py-1 rounded-md font-bold">✏️ 문구</button>

                      isReadyToPlace

                {isReadyToPlace && (

                    onClick={() => setIsReadyToPlace(false)}
                    className="text-xs text-center text-gray-500 underline hover:text-red-500"
                  >
                    취소하고 다시 적기

                    onClick={() => setIsCompleted(true)}

                  !isCompleted && isReadyToPlace ? 'cursor-crosshair ring-4 ring-blue-400' : 'cursor-default'

                  transformStyle: 'preserve-3d',
                  transform: isCompleted ? 'rotateX(42deg) rotateZ(-18deg) scale(0.88) translateY(-30px)' : 'rotateX(0deg) rotateZ(0deg) scale(1)',

                          left: '0', top: '0',
                          width: isCompleted ? '40px' : '0px',
                          height: isCompleted ? '12px' : '0px',

                          opacity: isCompleted ? 1 : 0,

                      {/* 건물 본체 (완성 시 지도 평면에서 수직으로 90도 일어남) */}
                      <div
                        className="relative flex flex-col items-center gap-0.5 hover:z-50 cursor-pointer"
                        style={{
                          transformOrigin: 'bottom center',
                          // rotateZ(18deg)로 지도의 회전을 상쇄해 정면을 보게 하고,
                          // rotateX(-90deg)를 주어 지도 바닥에서 꼿꼿이 세웁니다.
                          transform: isCompleted 
                            ? `translate(-50%, -100%) rotateZ(18deg) rotateX(-90deg) scale(1.3)` 
                            : `translate(-50%, -50%) scale(1)`,
                          transition: 'transform 1s cubic-bezier(0.34, 1.56, 0.64, 1)',
                          transformStyle: 'preserve-3d'
                        }}
                      >
                        {shop.profit && shop.profit !== '0' && (
                          <div className={`border-2 font-black px-3 py-0.5 rounded-full whitespace-nowrap transition-all duration-1000 ${
                            isCompleted ? 'bg-yellow-400 text-yellow-950 border-yellow-200 shadow-[0_4px_12px_rgba(250,204,21,0.6)] text-[10px]' : 'bg-white text-orange-600 border-orange-300 shadow-md text-[11px]'
                          }`}>
                            {shop.profit}
                          </div>
                        )}
                        
                        <div className="text-4xl md:text-5xl hover:scale-110 transition-transform drop-shadow-md">
                          {shop.icon}
                        </div>
                        
                        <div className={`text-[11px] md:text-xs font-black px-3 py-1.5 rounded-xl whitespace-nowrap border-2 transition-all duration-1000 ${
                          isCompleted ? 'bg-indigo-900 text-yellow-200 border-indigo-400 shadow-xl' : 'bg-gray-900 text-white border-gray-700 shadow-md'
                        }`}>
                          {shop.name}
                        </div>
                      </div>
                    </div>
                  ))}
                </div>
              </div>
            </div>
          </div>
        </div>
      );
    }

    const root = ReactDOM.createRoot(document.getElementById('root'));
    root.render(<App />);
  </script>
